// Pure, Mathematically Tuned Chinese Gong Synthesizer (6-second natural ring) function playGongSound() { try { const AudioContext = window.AudioContext || window.webkitAudioContext; const ctx = new AudioContext(); const now = ctx.currentTime; const totalDuration = 6.0; // Master Volume Control const masterGain = ctx.createGain(); masterGain.gain.setValueAtTime(0.6, now); masterGain.gain.exponentialRampToValueAtTime(0.0001, now + totalDuration); masterGain.connect(ctx.destination); // Harmonic ratios tuned to pure intervals (E2, A2, C#3, E3, A3, G#4) // Pure sine waves eliminate harsh, un-tuned distortion const frequencies = [82.41, 110.00, 138.59, 164.81, 220.00, 415.30]; frequencies.forEach((freq, i) => { const osc = ctx.createOscillator(); const gain = ctx.createGain(); osc.type = 'sine'; // Pure harmonic tone // Pitch strike simulation: Starts slightly sharp (1%) and settles immediately into true pitch osc.frequency.setValueAtTime(freq * 1.01, now); osc.frequency.exponentialRampToValueAtTime(freq, now + 0.08); // Attack & Decay Envelope // Fundamental pitches (low) decay slowly; higher shimmer tones decay faster const peakGain = 0.4 / (i + 1); const decayTime = totalDuration * (1 - (i * 0.12)); gain.gain.setValueAtTime(0, now); gain.gain.linearRampToValueAtTime(peakGain, now + 0.03); // Instant strike impact gain.gain.exponentialRampToValueAtTime(0.0001, now + decayTime); osc.connect(gain); gain.connect(masterGain); osc.start(now); osc.stop(now + totalDuration); }); } catch (e) { console.log("Audio waiting for user click.", e); } }
